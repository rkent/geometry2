77 void setEuler(const tf2Scalar& yaw, const tf2Scalar& pitch, const tf2Scalar& roll)
78 {

232 {
233 tf2Scalar s = tf2Sqrt(length2() * q.length2());
234 tf2Assert(s != tf2Scalar(0.0));

331 TF2_PUBLIC
332 Quaternion slerp(const Quaternion& q, const tf2Scalar& t) const
333 {
334 tf2Scalar theta = angleShortestPath(q) / tf2Scalar(2.0);
335 if (theta != tf2Scalar(0.0))
336 {
337 tf2Scalar d = tf2Scalar(1.0) / tf2Sin(theta);
338 tf2Scalar s0 = tf2Sin((tf2Scalar(1.0) - t) * theta);
339 tf2Scalar s1 = tf2Sin(t * theta);
340 if (dot(q) < 0) // Take care of long angle case see http://en.wikipedia.org/wiki/Slerp
341 return Quaternion((m_floats[0] * s0 + -q.x() * s1) * d,
342 (m_floats[1] * s0 + -q.y() * s1) * d,
343 (m_floats[2] * s0 + -q.z() * s1) * d,
344 (m_floats[3] * s0 + -q.m_floats[3] * s1) * d);
345 else
346 return Quaternion((m_floats[0] * s0 + q.x() * s1) * d,
347 (m_floats[1] * s0 + q.y() * s1) * d,
348 (m_floats[2] * s0 + q.z() * s1) * d,
349 (m_floats[3] * s0 + q.m_floats[3] * s1) * d);
350
351 }
352 else
353 {
354 return *this;
355 }

462TF2SIMD_FORCE_INLINE Quaternion
463shortestArcQuat(const Vector3& v0, const Vector3& v1) // Game Programming Gems 2.10. make sure v0,v1 are normalized
464{
465 Vector3 c = v0.cross(v1);
466 tf2Scalar d = v0.dot(v1);
467
468 if (d < -1.0 + TF2SIMD_EPSILON)
469 {
470 Vector3 n,unused;
471 tf2PlaneSpace1(v0,n,unused);
472 return Quaternion(n.x(),n.y(),n.z(),0.0f); // just pick any vector that is orthogonal to v0
473 }
474
475 tf2Scalar s = tf2Sqrt((1.0f + d) * 2.0f);
476 tf2Scalar rs = 1.0f / s;
477
478 return Quaternion(c.getX()*rs,c.getY()*rs,c.getZ()*rs,s * 0.5f);
479}
